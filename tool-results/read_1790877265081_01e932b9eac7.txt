   160→   160→function Highlight({ text, query }: { text: string; query: string }) {
   161→   161→  const q = query.trim();
   162→   162→  if (!q) return <>{text}</>;
   163→   163→  const escaped = q.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
   164→   164→  let parts: string[];
   165→   165→  try {
   166→   166→    parts = text.split(new RegExp(`(${escaped})`, "ig"));
   167→   167→  } catch {
   168→   168→    return <>{text}</>;
   169→   169→  }
   170→   170→  const lower = q.toLowerCase();
   171→   171→  return (
   172→   172→    <>
   173→   173→      {parts.map((part, i) =>
   174→   174→        part.toLowerCase() === lower ? (
   175→   175→          <mark key={i} className="rounded-[3px] bg-emerald-500/25 px-0.5 text-emerald-200">
   176→   176→            {part}
   177→   177→          </mark>
   178→   178→        ) : (
   179→   179→          <span key={i}>{part}</span>
   180→   180→        )
   181→   181→      )}
   182→   182→    </>
   183→   183→  );
   184→   184→}
