   127→function SeverityBadge({ severity }: { severity: string }) {
   128→  const map: Record<string, string> = {
   129→    crit: "border-red-500/50 bg-red-500/10 text-red-400",
   130→    warn: "border-amber-500/50 bg-amber-500/10 text-amber-400",
   131→    info: "border-slate-500/50 bg-slate-500/10 text-slate-300",
   132→  };
   133→  const label: Record<string, string> = { crit: "CRIT", warn: "WARN", info: "INFO" };
   134→  return (
   135→    <Badge variant="outline" className={`px-1.5 py-0 text-[10px] font-semibold ${map[severity] ?? map.info}`}>
   136→      {label[severity] ?? severity.toUpperCase()}
   137→    </Badge>
   138→  );
   139→}

   155→function PacketDetailSheet({
   156→  packet,
   157→  open,
   158→  onOpenChange,
   159→  detail,
   160→  loading,
   161→}: {
   162→  packet: Packet | null;
   163→  open: boolean;
   164→  onOpenChange: (v: boolean) => void;
   165→  detail: { hexdump: string } | null;
   166→  loading: boolean;
   167→}) {
   168→  const rows: [string, string][] = packet
   169→    ? [
   170→        ["ID", `#${packet.id}`],
   171→        ["Время", fmtTime(packet.ts)],
   172→        ["Направление", packet.direction],
   173→        ["Клиент", packet.client],
   174→        ["Канал", packet.portName],
   175→        ["Протокол", packet.protocol],
   176→        ["msg_type", packet.msgType ?? "—"],
   177→        ["method_type", packet.methodType ?? "—"],
   178→        ["method", packet.method ?? "—"],
   179→        ["cmd_type", packet.cmdType ?? "—"],
   180→        ["cmd_class / cmd", packet.cmdClass != null ? `${packet.cmdClass} / ${packet.cmd ?? "—"}` : "—"],
   181→        ["Размер", fmtBytes(packet.size)],
   182→        ["Валидность", packet.valid ? "✓ корректный" : "✗ повреждён/некорректен"],
   183→      ]
   184→    : [];
   185→
   186→  return (
   187→    <Sheet open={open} onOpenChange={onOpenChange}>
   188→      <SheetContent
   189→        side="right"
   190→        className="w-full max-w-md overflow-y-auto border-slate-800 bg-slate-950 p-4 sm:max-w-lg custom-scroll"
   191→      >
   192→        <SheetHeader className="p-0 pb-2 text-left">
   193→          <SheetTitle className="text-base text-slate-100">
   194→            Пакет {packet ? `#${packet.id}` : ""}
   195→          </SheetTitle>
   196→          <SheetDescription className="text-xs text-slate-400">
   197→            {packet ? packet.summary : ""}
   198→          </SheetDescription>
   199→        </SheetHeader>
   200→        {packet && (
   201→          <div className="mt-2 space-y-3">
   202→            <div className="rounded-lg border border-slate-800">
   203→              <Table>
   204→                <TableBody>
   205→                  {rows.map(([k, v]) => (
   206→                    <TableRow key={k} className="border-slate-800/70">
   207→                      <TableCell className="py-1.5 pl-3 pr-2 font-mono text-[11px] text-slate-500">
   208→                        {k}
   209→                      </TableCell>
   210→                      <TableCell className="py-1.5 pr-3 font-mono text-xs text-slate-200">
   211→                        {v}
   212→                      </TableCell>
   213→                    </TableRow>
   214→                  ))}
   215→                </TableBody>
   216→              </Table>
   217→            </div>
   218→            <div>
   219→              <p className="mb-1 text-[11px] font-semibold uppercase tracking-wider text-slate-500">
   220→                Hexdump {loading ? "" : detail ? "(до 2 КБ)" : ""}
   221→              </p>
   222→              {loading ? (
   223→                <div className="flex items-center gap-2 rounded-lg border border-slate-800 bg-slate-950 p-4 text-xs text-slate-400">
   224→                  <Loader2 className="size-4 animate-spin" aria-hidden /> Загрузка дампа…
   225→                </div>
   226→              ) : (
   227→                <pre className="max-h-72 overflow-auto rounded-lg border border-slate-800 bg-slate-950 p-3 font-mono text-[10px] leading-relaxed text-emerald-300 custom-scroll">
   228→                  {detail?.hexdump ?? packet.hexPreview}
   229→                </pre>
   230→              )}
   231→            </div>
   232→          </div>
   233→        )}
   234→      </SheetContent>
   235→    </Sheet>
   236→  );
   237→}

   239→export function SnifferTab({ live }: SnifferTabProps) {
   240→  const { connected, packets, sessions, alerts, stats, series, scenario } = live;
   241→
   242→  // Фильтры
   243→  const [search, setSearch] = useState("");
   244→  const [protocol, setProtocol] = useState("all");
   245→  const [direction, setDirection] = useState("all");
   246→  const [minSize, setMinSize] = useState("");
   247→  const [maxSize, setMaxSize] = useState("");
   248→  const [paused, setPaused] = useState(false);
   249→  const [frozenPackets, setFrozenPackets] = useState<Packet[] | null>(null);
   250→  const [muted, setMuted] = useState(false);
   251→  const [selected, setSelected] = useState<Packet | null>(null);
   252→  const [sheetOpen, setSheetOpen] = useState(false);
   253→  const [detail, setDetail] = useState<{ hexdump: string } | null>(null);
   254→  const [detailLoading, setDetailLoading] = useState(false);
   255→  const [scenarioBusy, setScenarioBusy] = useState<ScenarioId | "stop" | null>(null);
   256→  const [nowTick, setNowTick] = useState(() => Date.now());
   257→  const alertToastSeenRef = useRef<string | null>(null);
   258→
   259→  // Тосты по живым алертам (если не в mute) — реагируем только на новые id
   260→  useEffect(() => {
   261→    if (muted || alerts.length === 0) return;
   262→    const latest = alerts[0];
   263→    if (alertToastSeenRef.current === latest.id) return;
   264→    const t = new Date(latest.createdAt).getTime();
   265→    if (Date.now() - t < 2500) {
   266→      alertToastSeenRef.current = latest.id;
   267→      if (latest.severity === "crit") {
   268→        toast.error(latest.rule, { description: latest.message, duration: 3500 });
   269→      } else {
   270→        toast.warning(latest.rule, { description: latest.message, duration: 3000 });
   271→      }
   272→    }
   273→  }, [alerts, muted]);
   274→
   275→  // Тикаем часы для обратного отсчёта сценария
   276→  useEffect(() => {
   277→    if (!scenario) return;
   278→    const timer = setInterval(() => setNowTick(Date.now()), 1000);
   279→    return () => clearInterval(timer);
   280→  }, [scenario]);
   281→
   282→  const scenarioRemaining = useMemo(() => {
   283→    if (!scenario) return 0;
   284→    return Math.max(0, Math.ceil((new Date(scenario.endsAt).getTime() - nowTick) / 1000));
   285→  }, [scenario, nowTick]);
   286→
   287→  const handleScenario = async (id: ScenarioId | "stop") => {
   288→    setScenarioBusy(id);
   289→    try {
   290→      const res =
   291→        id === "stop"
   292→          ? await fetch("/api/sniffer/scenario", { method: "DELETE" })
   293→          : await fetch("/api/sniffer/scenario", {
   294→              method: "POST",
   295→              headers: { "Content-Type": "application/json" },
   296→              body: JSON.stringify({ id }),
   297→            });
   298→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   299→      if (id === "stop") {
   300→        toast("Сценарий остановлен", { duration: 2000 });
   301→      } else {
   302→        const def = SCENARIO_CATALOG.find((s) => s.id === id);
   303→        toast.success(`Сценарий «${def?.label}» запущен`, {
   304→          description: "Генератор трафика изменил профиль — следите за графиком и тревогами.",
   305→          duration: 3000,
   306→        });
   307→      }
   308→    } catch (e) {
   309→      toast.error("Не удалось изменить сценарий", {
   310→        description: e instanceof Error ? e.message : String(e),
   311→        duration: 3000,
   312→      });
   313→    } finally {
   314→      setScenarioBusy(null);
   315→    }
   316→  };
   317→
   318→  const handlePause = () => {
   319→    if (!paused) {
   320→      setFrozenPackets(packets);
   321→      setPaused(true);
   322→      toast.info("Таблица остановлена", {
   323→        description: "Приём пакетов продолжается в фоне (буфер 3000).",
   324→        duration: 2500,
   325→      });
   326→    } else {
   327→      setFrozenPackets(null);
   328→      setPaused(false);
   329→    }
   330→  };
   331→
   332→  const filtered = useMemo(() => {
   333→    const source = paused && frozenPackets ? frozenPackets : packets;
   334→    const s = search.trim().toLowerCase();
   335→    const min = parseInt(minSize, 10);
   336→    const max = parseInt(maxSize, 10);
   337→    return source.filter((p) => {
   338→      if (protocol !== "all" && p.protocol !== protocol) return false;
   339→      if (direction !== "all" && p.direction !== direction) return false;
   340→      if (!Number.isNaN(min) && p.size < min) return false;
   341→      if (!Number.isNaN(max) && p.size > max) return false;
   342→      if (s) {
   343→        const hay = `${p.summary} ${p.method ?? ""} ${p.methodType ?? ""} ${p.cmdType ?? ""} ${p.client}`.toLowerCase();
   344→        if (!hay.includes(s)) return false;
   345→      }
   346→      return true;
   347→    });
   348→  }, [packets, frozenPackets, paused, search, protocol, direction, minSize, maxSize]);
   349→
   350→  const display = filtered.slice(0, MAX_TABLE_ROWS);
   351→  const latestAlerts = alerts.slice(0, 50);
   352→  const sessionList = sessions.slice(0, 12);
   353→
   354→  const onRowClick = (p: Packet) => {
   355→    setSelected(p);
   356→    setSheetOpen(true);
   357→    setDetail(null);
   358→    setDetailLoading(true);
   359→    fetch(`/api/sniffer/packet/${p.id}`)
   360→      .then((r) => (r.ok ? r.json() : Promise.reject(new Error("404"))))
   361→      .then((d: { hexdump: string }) => setDetail(d))
   362→      .catch(() => setDetail({ hexdump: p.hexPreview }))
   363→      .finally(() => setDetailLoading(false));
   364→  };
   365→
   366→  return (
   367→    <div className="space-y-4">
   368→      {/* Статбар */}
   369→      <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">
   370→        <StatCard
   371→          icon={Activity}
   372→          label="Пакеты всего"
   373→          value={stats ? fmtCompact(stats.totalPackets) : "—"}
   374→          sub={stats ? `${stats.lastPps} п/с · ${stats.invalidPackets} невалидных` : undefined}
   375→        />
   376→        <StatCard
   377→          icon={HardDrive}
   378→          label="Байт перехвачено"
   379→          value={stats ? fmtBytes(stats.totalBytes) : "—"}
   380→          sub={stats ? `${fmtCompact(stats.lastBps)} Б/с` : undefined}
   381→        />
   382→        <StatCard
   383→          icon={Network}
   384→          label="Активные сессии"
   385→          value={stats ? String(stats.activeSessions) : "—"}
   386→          sub={stats ? `всего за сессию: ${stats.totalSessions}` : undefined}
   387→        />
   388→        <StatCard
   389→          icon={Siren}
   390→          label="Тревоги"
   391→          value={stats ? String(stats.alertsCount) : "—"}
   392→          sub="правила config.json"
   393→          alert={(stats?.alertsCount ?? 0) > 0}
   394→        />
   395→      </div>
   396→
   397→      {/* Сценарии трафика (демо-режимы генератора) */}
   398→      <Card
   399→        className={`border-slate-800 bg-slate-900/60 ${
   400→          scenario ? "border-amber-500/50 shadow-[0_0_18px_-6px] shadow-amber-500/30" : ""
   401→        }`}
   402→      >
   403→        <CardHeader className="flex-row flex-wrap items-center justify-between gap-2 p-4 pb-2">
   404→          <CardTitle className="flex items-center gap-2 text-sm text-slate-200">
   405→            <Zap className={`size-4 ${scenario ? "text-amber-400" : "text-emerald-400"}`} aria-hidden />
   406→            Сценарии трафика
   407→            {scenario && (
   408→              <Badge
   409→                variant="outline"
   410→                className="border-amber-500/50 bg-amber-500/10 font-mono text-[10px] text-amber-300"
   411→              >
   412→                {scenario.label} · осталось {scenarioRemaining}с
   413→              </Badge>
   414→            )}
   415→          </CardTitle>
   416→          <span className="text-[11px] text-slate-500">
   417→            демо-режимы генератора — как если бы АЗС начали массовый опрос
   418→          </span>
   419→        </CardHeader>
   420→        <CardContent className="flex flex-wrap gap-2 p-4 pt-0">
   421→          {SCENARIO_CATALOG.map((s) => {
   422→            const isActive = scenario?.id === s.id;
   423→            const busy = scenarioBusy === s.id;
   424→            return (
   425→              <Button
   426→                key={s.id}
   427→                size="sm"
   428→                variant="outline"
   429→                disabled={scenarioBusy !== null}
   430→                onClick={() => handleScenario(s.id)}
   431→                aria-pressed={isActive}
   432→                className={`min-h-[44px] flex-col items-start gap-0 border-slate-700 sm:min-h-[40px] ${
   433→                  isActive
   434→                    ? "border-amber-500/60 bg-amber-500/15 text-amber-200 hover:bg-amber-500/25"
   435→                    : "bg-slate-950 text-slate-300 hover:bg-slate-800 hover:text-slate-100"
   436→                }`}
   437→              >
   438→                <span className="flex items-center gap-1.5 text-xs font-medium">
   439→                  {busy && <Loader2 className="size-3 animate-spin" aria-hidden />}
   440→                  {s.label}
   441→                </span>
   442→                <span className="font-mono text-[10px] text-slate-500">{s.hint}</span>
   443→              </Button>
   444→            );
   445→          })}
   446→          {scenario && (
   447→            <Button
   448→              size="sm"
   449→              variant="outline"
   450→              disabled={scenarioBusy !== null}
   451→              onClick={() => handleScenario("stop")}
   452→              className="ml-auto min-h-[44px] border-red-500/50 bg-red-500/10 text-red-300 hover:bg-red-500/20 hover:text-red-200 sm:min-h-[40px]"
   453→            >
   454→              {scenarioBusy === "stop" ? (
   455→                <Loader2 className="size-4 animate-spin" aria-hidden />
   456→              ) : (
   457→                <Square className="size-3.5" aria-hidden />
   458→              )}
   459→              Остановить
   460→            </Button>
   461→          )}
   462→        </CardContent>
   463→      </Card>
   464→
   465→      {/* График */}
   466→      <Card className="border-slate-800 bg-slate-900/60">
   467→        <CardHeader className="p-4 pb-2">
   468→          <CardTitle className="flex flex-wrap items-center justify-between gap-2 text-sm text-slate-200">
   469→            <span>Пакетов в секунду (pps)</span>
   470→            {stats && (
   471→              <Badge variant="outline" className="border-slate-700 font-mono text-[10px] text-slate-400">
   472→                uptime {Math.floor(stats.uptimeSec / 60)}м {stats.uptimeSec % 60}с
   473→              </Badge>
   474→            )}
   475→          </CardTitle>
   476→        </CardHeader>
   477→        <CardContent className="px-4 pb-4 pt-0">
   478→          <PpsChart series={series} />
   479→        </CardContent>
   480→      </Card>
   481→
   482→      {/* Распределение протоколов + топы */}
   483→      {stats && (
   484→        <Card className="border-slate-800 bg-slate-900/60">
   485→          <CardHeader className="p-4 pb-3">
   486→            <CardTitle className="text-sm text-slate-200">Аналитика трафика</CardTitle>
   487→          </CardHeader>
   488→          <CardContent className="p-4 pt-0">
   489→            <ProtocolBreakdown
   490→              perProtocol={stats.perProtocol}
   491→              totalPackets={stats.totalPackets}
   492→              topMethods={stats.topMethods}
   493→              topClients={stats.topClients}
   494→            />
   495→          </CardContent>
   496→        </Card>
   497→      )}
   498→
   499→      {/* Фильтры */}
   500→      <Card className="border-slate-800 bg-slate-900/60">
   501→        <CardContent className="grid gap-3 p-4 md:grid-cols-[1fr_auto_auto_auto_auto] lg:grid-cols-[1.4fr_auto_auto_auto_auto_auto]">
   502→          <div className="space-y-1">
   503→            <Label htmlFor="pkt-search" className="text-[11px] text-slate-400">
   504→              Поиск (метод, резюме, клиент)
   505→            </Label>
   506→            <Input
   507→              id="pkt-search"
   508→              placeholder="напр. sendTanksState или 10.8.0.51"
