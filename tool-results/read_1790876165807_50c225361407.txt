   140→function SeverityBadge({ severity }: { severity: string }) {
   141→  const map: Record<string, string> = {
   142→    crit: "border-red-500/50 bg-red-500/10 text-red-400",
   143→    warn: "border-amber-500/50 bg-amber-500/10 text-amber-400",
   144→    info: "border-slate-500/50 bg-slate-500/10 text-slate-300",
   145→  };
   146→  const label: Record<string, string> = { crit: "CRIT", warn: "WARN", info: "INFO" };
   147→  return (
   148→    <Badge variant="outline" className={`px-1.5 py-0 text-[10px] font-semibold ${map[severity] ?? map.info}`}>
   149→      {label[severity] ?? severity.toUpperCase()}
   150→    </Badge>
   151→  );
   152→}

   168→function PacketDetailSheet({
   169→  packet,
   170→  pendingId,
   171→  open,
   172→  onOpenChange,
   173→  detail,
   174→  loading,
   175→}: {
   176→  packet: Packet | null;
   177→  pendingId: number | null;
   178→  open: boolean;
   179→  onOpenChange: (v: boolean) => void;
   180→  detail: { hexdump: string } | null;
   181→  loading: boolean;
   182→}) {
   183→  const [copied, setCopied] = useState(false);
   184→
   185→  const rows: [string, string][] = packet
   186→    ? [
   187→        ["ID", `#${packet.id}`],
   188→        ["Время", fmtTime(packet.ts)],
   189→        ["Направление", packet.direction],
   190→        ["Клиент", packet.client],
   191→        ["Канал", packet.portName],
   192→        ["Протокол", packet.protocol],
   193→        ["msg_type", packet.msgType ?? "—"],
   194→        ["method_type", packet.methodType ?? "—"],
   195→        ["method", packet.method ?? "—"],
   196→        ["cmd_type", packet.cmdType ?? "—"],
   197→        ["cmd_class / cmd", packet.cmdClass != null ? `${packet.cmdClass} / ${packet.cmd ?? "—"}` : "—"],
   198→        ["Размер", fmtBytes(packet.size)],
   199→        ["Валидность", packet.valid ? "✓ корректный" : "✗ повреждён/некорректен"],
   200→      ]
   201→    : [];
   202→
   203→  const hexdumpText = detail?.hexdump ?? packet?.hexPreview ?? "";
   204→
   205→  const handleCopy = async () => {
   206→    if (!hexdumpText) return;
   207→    try {
   208→      await navigator.clipboard.writeText(hexdumpText);
   209→      setCopied(true);
   210→      setTimeout(() => setCopied(false), 1500);
   211→    } catch {
   212→      /* clipboard недоступен */
   213→    }
   214→  };
   215→
   216→  return (
   217→    <Sheet open={open} onOpenChange={onOpenChange}>
   218→      <SheetContent
   219→        side="right"
   220→        className="w-full max-w-md overflow-y-auto border-slate-800 bg-slate-950 p-4 sm:max-w-lg custom-scroll"
   221→      >
   222→        <SheetHeader className="p-0 pb-2 text-left">
   223→          <SheetTitle className="text-base text-slate-100">
   224→            Пакет {packet ? `#${packet.id}` : pendingId ? `#${pendingId}` : ""}
   225→          </SheetTitle>
   226→          <SheetDescription className="text-xs text-slate-400">
   227→            {packet ? packet.summary : pendingId ? "Загрузка пакета из буфера сервера…" : ""}
   228→          </SheetDescription>
   229→        </SheetHeader>
   230→        {packet ? (
   231→          <div className="mt-2 space-y-3">
   232→            <div className="rounded-lg border border-slate-800">
   233→              <Table>
   234→                <TableBody>
   235→                  {rows.map(([k, v]) => (
   236→                    <TableRow key={k} className="border-slate-800/70">
   237→                      <TableCell className="py-1.5 pl-3 pr-2 font-mono text-[11px] text-slate-500">
   238→                        {k}
   239→                      </TableCell>
   240→                      <TableCell className="py-1.5 pr-3 font-mono text-xs text-slate-200">
   241→                        {v}
   242→                      </TableCell>
   243→                    </TableRow>
   244→                  ))}
   245→                </TableBody>
   246→              </Table>
   247→            </div>
   248→            <div>
   249→              <div className="mb-1 flex items-center justify-between">
   250→                <p className="text-[11px] font-semibold uppercase tracking-wider text-slate-500">
   251→                  Hexdump {loading ? "" : detail ? "(до 2 КБ)" : ""}
   252→                </p>
   253→                {!loading && hexdumpText && (
   254→                  <Button
   255→                    size="sm"
   256→                    variant="ghost"
   257→                    onClick={handleCopy}
   258→                    className="h-7 gap-1 px-2 text-[11px] text-slate-400 hover:bg-slate-800 hover:text-emerald-300"
   259→                    aria-label="Скопировать hexdump"
   260→                  >
   261→                    <Copy className="size-3" aria-hidden />
   262→                    {copied ? "скопировано" : "копировать"}
   263→                  </Button>
   264→                )}
   265→              </div>
   266→              {loading ? (
   267→                <div className="flex items-center gap-2 rounded-lg border border-slate-800 bg-slate-950 p-4 text-xs text-slate-400">
   268→                  <Loader2 className="size-4 animate-spin" aria-hidden /> Загрузка дампа…
   269→                </div>
   270→              ) : (
   271→                <pre className="max-h-72 overflow-auto rounded-lg border border-slate-800 bg-slate-950 p-3 font-mono text-[10px] leading-relaxed text-emerald-300 custom-scroll">
   272→                  {hexdumpText}
   273→                </pre>
   274→              )}
   275→            </div>
   276→          </div>
   277→        ) : pendingId ? (
   278→          <div className="mt-4 flex items-center gap-2 rounded-lg border border-slate-800 bg-slate-950 p-4 text-xs text-slate-400">
   279→            <Loader2 className="size-4 animate-spin" aria-hidden /> Запрос GET /api/sniffer/packet/{pendingId}…
   280→          </div>
   281→        ) : null}
   282→      </SheetContent>
   283→    </Sheet>
   284→  );
   285→}

   287→export function SnifferTab({ live }: SnifferTabProps) {
   288→  const { connected, packets, sessions, alerts, stats, series, scenario } = live;
   289→
   290→  // Фильтры
   291→  const [search, setSearch] = useState("");
   292→  const [protocol, setProtocol] = useState("all");
   293→  const [direction, setDirection] = useState("all");
   294→  const [minSize, setMinSize] = useState("");
   295→  const [maxSize, setMaxSize] = useState("");
   296→  const [paused, setPaused] = useState(false);
   297→  const [frozenPackets, setFrozenPackets] = useState<Packet[] | null>(null);
   298→  const [muted, setMuted] = useState(false);
   299→  const [selected, setSelected] = useState<Packet | null>(null);
   300→  const [pendingId, setPendingId] = useState<number | null>(null);
   301→  const [sheetOpen, setSheetOpen] = useState(false);
   302→  const [detail, setDetail] = useState<{ hexdump: string } | null>(null);
   303→  const [detailLoading, setDetailLoading] = useState(false);
   304→  const [metric, setMetric] = useState<ChartMetric>("pps");
   305→  const [scenarioBusy, setScenarioBusy] = useState<ScenarioId | "stop" | null>(null);
   306→  const [nowTick, setNowTick] = useState(() => Date.now());
   307→  const searchInputRef = useRef<HTMLInputElement>(null);
   308→  const tableCardRef = useRef<HTMLDivElement>(null);
   309→  const alertToastSeenRef = useRef<string | null>(null);
   310→
   311→  // Тосты по живым алертам (если не в mute) — реагируем только на новые id
   312→  useEffect(() => {
   313→    if (muted || alerts.length === 0) return;
   314→    const latest = alerts[0];
   315→    if (alertToastSeenRef.current === latest.id) return;
   316→    const t = new Date(latest.createdAt).getTime();
   317→    if (Date.now() - t < 2500) {
   318→      alertToastSeenRef.current = latest.id;
   319→      if (latest.severity === "crit") {
   320→        toast.error(latest.rule, { description: latest.message, duration: 3500 });
   321→      } else {
   322→        toast.warning(latest.rule, { description: latest.message, duration: 3000 });
   323→      }
   324→    }
   325→  }, [alerts, muted]);
   326→
   327→  // Тикаем часы для обратного отсчёта сценария
   328→  useEffect(() => {
   329→    if (!scenario) return;
   330→    const timer = setInterval(() => setNowTick(Date.now()), 1000);
   331→    return () => clearInterval(timer);
   332→  }, [scenario]);
   333→
   334→  const scenarioRemaining = useMemo(() => {
   335→    if (!scenario) return 0;
   336→    return Math.max(0, Math.ceil((new Date(scenario.endsAt).getTime() - nowTick) / 1000));
   337→  }, [scenario, nowTick]);
   338→
   339→  const handleScenario = async (id: ScenarioId | "stop") => {
   340→    setScenarioBusy(id);
   341→    try {
   342→      const res =
   343→        id === "stop"
   344→          ? await fetch("/api/sniffer/scenario", { method: "DELETE" })
   345→          : await fetch("/api/sniffer/scenario", {
   346→              method: "POST",
   347→              headers: { "Content-Type": "application/json" },
   348→              body: JSON.stringify({ id }),
   349→            });
   350→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   351→      if (id === "stop") {
   352→        toast("Сценарий остановлен", { duration: 2000 });
   353→      } else {
   354→        const def = SCENARIO_CATALOG.find((s) => s.id === id);
   355→        toast.success(`Сценарий «${def?.label}» запущен`, {
   356→          description: "Генератор трафика изменил профиль — следите за графиком и тревогами.",
   357→          duration: 3000,
   358→        });
   359→      }
   360→    } catch (e) {
   361→      toast.error("Не удалось изменить сценарий", {
   362→        description: e instanceof Error ? e.message : String(e),
   363→        duration: 3000,
   364→      });
   365→    } finally {
   366→      setScenarioBusy(null);
   367→    }
   368→  };
   369→
   370→  const handlePause = () => {
   371→    if (!paused) {
   372→      setFrozenPackets(packets);
   373→      setPaused(true);
   374→      toast.info("Таблица остановлена", {
   375→        description: "Приём пакетов продолжается в фоне (буфер 3000).",
   376→        duration: 2500,
   377→      });
   378→    } else {
   379→      setFrozenPackets(null);
   380→      setPaused(false);
   381→    }
   382→  };
   383→
   384→  // Горячие клавиши вкладки: P — пауза, / — фокус на поиск
   385→  useEffect(() => {
   386→    const onKey = (e: KeyboardEvent) => {
   387→      const t = e.target as HTMLElement | null;
   388→      const typing =
   389→        !!t && (t.tagName === "INPUT" || t.tagName === "TEXTAREA" || t.tagName === "SELECT" || t.isContentEditable);
   390→      if (e.key.toLowerCase() === "p" || e.key === "з") {
   391→        if (typing || sheetOpen) return;
   392→        e.preventDefault();
   393→        handlePause();
   394→      } else if (e.key === "/") {
   395→        if (sheetOpen) return;
   396→        e.preventDefault();
   397→        searchInputRef.current?.focus();
   398→      }
   399→    };
   400→    window.addEventListener("keydown", onKey);
   401→    return () => window.removeEventListener("keydown", onKey);
   402→  }, [sheetOpen, packets, paused]);
   403→
   404→  const openPacketById = (id: number) => {
   405→    const local = packets.find((p) => p.id === id);
   406→    if (local) {
   407→      onRowClick(local);
   408→      return;
   409→    }
   410→    setPendingId(id);
   411→    setSelected(null);
   412→    setSheetOpen(true);
   413→    setDetail(null);
   414→    setDetailLoading(true);
   415→    fetch(`/api/sniffer/packet/${id}`)
   416→      .then((r) => (r.ok ? r.json() : Promise.reject(new Error(r.status === 404 ? "notfound" : String(r.status)))))
   417→      .then((d: { packet: Packet; hexdump: string }) => {
   418→        setSelected(d.packet);
   419→        setDetail({ hexdump: d.hexdump });
   420→      })
   421→      .catch(() => {
   422→        toast.error(`Пакет #${id} недоступен`, {
   423→          description: "Вытеснен из кольцевого буфера сниффера (5000 пакетов).",
   424→          duration: 3000,
   425→        });
   426→        setSheetOpen(false);
   427→      })
   428→      .finally(() => {
   429→        setPendingId(null);
   430→        setDetailLoading(false);
   431→      });
   432→  };
   433→
   434→  const filterByClient = (clientAddr: string) => {
   435→    const ip = clientAddr.split(":")[0];
   436→    setSearch(ip);
   437→    setPaused(false);
   438→    setFrozenPackets(null);
   439→    tableCardRef.current?.scrollIntoView({ behavior: "smooth", block: "start" });
   440→    toast(`Фильтр по клиенту ${ip}`, {
   441→      description: "Таблица пакетов отфильтрована по адресам этого клиента.",
   442→      duration: 2500,
   443→    });
   444→  };
   445→
   446→  const filtered = useMemo(() => {
   447→    const source = paused && frozenPackets ? frozenPackets : packets;
   448→    const s = search.trim().toLowerCase();
   449→    const min = parseInt(minSize, 10);
   450→    const max = parseInt(maxSize, 10);
   451→    return source.filter((p) => {
   452→      if (protocol !== "all" && p.protocol !== protocol) return false;
   453→      if (direction !== "all" && p.direction !== direction) return false;
   454→      if (!Number.isNaN(min) && p.size < min) return false;
   455→      if (!Number.isNaN(max) && p.size > max) return false;
   456→      if (s) {
   457→        const hay = `${p.summary} ${p.method ?? ""} ${p.methodType ?? ""} ${p.cmdType ?? ""} ${p.client}`.toLowerCase();
   458→        if (!hay.includes(s)) return false;
   459→      }
   460→      return true;
   461→    });
   462→  }, [packets, frozenPackets, paused, search, protocol, direction, minSize, maxSize]);
   463→
   464→  const display = filtered.slice(0, MAX_TABLE_ROWS);
   465→  const latestAlerts = alerts.slice(0, 50);
   466→  const sessionList = sessions.slice(0, 12);
   467→
   468→  // Активные фильтры → чипы с быстрым сбросом
   469→  const chips = useMemo(() => {
   470→    const out: { key: string; label: string; clear: () => void }[] = [];
   471→    if (search.trim()) out.push({ key: "search", label: `поиск: ${search.trim()}`, clear: () => setSearch("") });
   472→    if (protocol !== "all") out.push({ key: "proto", label: `протокол: ${protocol}`, clear: () => setProtocol("all") });
   473→    if (direction !== "all") out.push({ key: "dir", label: direction === "TX" ? "только TX" : "только RX", clear: () => setDirection("all") });
   474→    if (minSize) out.push({ key: "min", label: `≥ ${minSize} Б`, clear: () => setMinSize("") });
   475→    if (maxSize) out.push({ key: "max", label: `≤ ${maxSize} Б`, clear: () => setMaxSize("") });
   476→    return out;
   477→  }, [search, protocol, direction, minSize, maxSize]);
   478→
   479→  const onRowClick = (p: Packet) => {
   480→    setSelected(p);
   481→    setSheetOpen(true);
   482→    setDetail(null);
   483→    setDetailLoading(true);
   484→    fetch(`/api/sniffer/packet/${p.id}`)
   485→      .then((r) => (r.ok ? r.json() : Promise.reject(new Error("404"))))
   486→      .then((d: { hexdump: string }) => setDetail(d))
   487→      .catch(() => setDetail({ hexdump: p.hexPreview }))
   488→      .finally(() => setDetailLoading(false));
   489→  };
   490→
   491→  return (
   492→    <div className="space-y-4">
   493→      {/* Статбар */}
   494→      <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">
   495→        <StatCard
   496→          icon={Activity}
   497→          label="Пакеты всего"
   498→          value={stats ? fmtCompact(stats.totalPackets) : "—"}
   499→          sub={stats ? `${stats.lastPps} п/с · ${stats.invalidPackets} невалидных` : undefined}
   500→          spark={<Sparkline data={series.map((s) => s.pps)} stroke="#10b981" />}
   501→        />
   502→        <StatCard
   503→          icon={HardDrive}
   504→          label="Байт перехвачено"
