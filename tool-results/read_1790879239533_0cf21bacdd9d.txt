   188→/** Подсветка совпадений поиска (без мутаций — чистый split по regex). */
   189→function Highlight({ text, query }: { text: string; query: string }) {
   190→  const q = query.trim();
   191→  if (!q) return <>{text}</>;
   192→  const escaped = q.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
   193→  let parts: string[];
   194→  try {
   195→    parts = text.split(new RegExp(`(${escaped})`, "ig"));
   196→  } catch {
   197→    return <>{text}</>;
   198→  }
   199→  const lower = q.toLowerCase();
   200→  return (
   201→    <>
   202→      {parts.map((part, i) =>
   203→        part.toLowerCase() === lower ? (
   204→          <mark key={i} className="rounded-[3px] bg-emerald-500/25 px-0.5 text-emerald-200">
   205→            {part}
   206→          </mark>
   207→        ) : (
   208→          <span key={i}>{part}</span>
   209→        )
   210→      )}
   211→    </>
   212→  );
   213→}

   253→/** Тренд KPI: среднее последних 60 с против предыдущих 60 с (из series). */
   254→interface Trend {
   255→  pct: number;
   256→  dir: "up" | "down" | "flat";
   257→}
   258→
   259→function computeTrend(values: number[]): Trend | null {
   260→  if (values.length < 70) return null;
   261→  const last = values.slice(-60);
   262→  const prev = values.slice(-120, -60);
   263→  if (prev.length < 30) return null;
   264→  const avg = (arr: number[]) => arr.reduce((a, b) => a + b, 0) / arr.length;
   265→  const now = avg(last);
   266→  const before = avg(prev);
   267→  if (before < 1e-9) return null;
   268→  const pct = ((now - before) / before) * 100;
   269→  if (Math.abs(pct) < 3) return { pct, dir: "flat" };
   270→  return { pct, dir: pct > 0 ? "up" : "down" };
   271→}

   285→      className={`hidden shrink-0 animate-in items-center gap-0.5 rounded border px-1 py-px font-mono text-[9px] leading-none fade-in duration-300 sm:inline-flex ${cfg.cls}`}

   306→/** Сравнение двух ручных срезов: дельты счётчиков + рост по протоколам (violet-акцент). */
   307→function SnapshotDiff({ a, b, onClear }: { a: Snapshot; b: Snapshot; onClear: () => void }) {
   308→  const diff = useMemo(() => {
   309→    const perProtocol: Record<string, number> = {};
   310→    for (const [proto, now] of Object.entries(b.perProtocol)) {
   311→      const delta = now - (a.perProtocol[proto] ?? 0);
   312→      if (delta > 0) perProtocol[proto] = delta;
   313→    }
   314→    const durationSec = Math.max(
   315→      1,
   316→      Math.round((new Date(b.ts).getTime() - new Date(a.ts).getTime()) / 1000)
   317→    );
   318→    return {
   319→      durationSec,
   320→      packets: Math.max(0, b.totalPackets - a.totalPackets),
   321→      bytes: Math.max(0, b.totalBytes - a.totalBytes),
   322→      alerts: Math.max(0, b.alerts - a.alerts),
   323→      invalid: Math.max(0, b.invalid - a.invalid),
   324→      perProtocol: Object.entries(perProtocol).sort((x, y) => y[1] - x[1]),
   325→    };
   326→  }, [a, b]);
   327→  const protoTotal = diff.perProtocol.reduce((acc, [, n]) => acc + n, 0);
   328→  const pps = Math.round((diff.packets / diff.durationSec) * 10) / 10;
   329→  return (
   330→    <div
   331→      role="status"
   332→      aria-label={`Сравнение срезов аналитики за ${diff.durationSec} с`}
   333→      className="mt-3 animate-in rounded-lg border border-violet-500/25 bg-violet-500/[0.04] p-3 fade-in slide-in-from-bottom-2 duration-300"
   334→    >
   335→      <div className="flex flex-wrap items-center justify-between gap-2">
   336→        <span className="flex items-center gap-2 text-xs font-medium text-slate-200">
   337→          <GitCompareArrows className="size-3.5 text-violet-400" aria-hidden />
   338→          Сравнение срезов A → B
   339→          <Badge
   340→            variant="outline"
   341→            className="border-slate-700 px-1.5 py-0 font-mono text-[10px] text-slate-400"
   342→          >
   343→            {fmtUptime(diff.durationSec)}
   344→          </Badge>
   345→        </span>
   346→        <button
   347→          type="button"
   348→          onClick={onClear}
   349→          aria-label="Сбросить сравнение срезов"
   350→          title="Сбросить срезы"
   351→          className="flex size-6 items-center justify-center rounded-md text-slate-500 hover:bg-slate-800 hover:text-slate-200"
   352→        >
   353→          <X className="size-3.5" aria-hidden />
   354→        </button>
   355→      </div>
   356→      <div className="mt-2.5 flex flex-wrap gap-x-4 gap-y-1.5">
   357→        <span className="inline-flex items-center gap-1.5 font-mono text-[11px] text-slate-300">
   358→          <Activity className="size-3 text-violet-400" aria-hidden />
   359→          <span className="font-semibold text-violet-300 tabular-nums">+{fmtCompact(diff.packets)}</span> пакетов
   360→          <span className="text-slate-500">· {pps}/с</span>
   361→        </span>
   362→        <span className="inline-flex items-center gap-1.5 font-mono text-[11px] text-slate-300">
   363→          <HardDrive className="size-3 text-cyan-400" aria-hidden />
   364→          <span className="font-semibold text-cyan-300 tabular-nums">+{fmtBytes(diff.bytes)}</span>
   365→        </span>
   366→        <span className="inline-flex items-center gap-1.5 font-mono text-[11px] text-slate-300">
   367→          <Siren className="size-3 text-red-400" aria-hidden />
   368→          <span className="font-semibold text-red-300 tabular-nums">+{diff.alerts}</span> тревог
   369→        </span>
   370→        {diff.invalid > 0 && (
   371→          <span className="inline-flex items-center gap-1.5 font-mono text-[11px] text-slate-400">
   372→            <XCircle className="size-3 text-amber-400" aria-hidden />
   373→            <span className="tabular-nums">{diff.invalid}</span> невалидных
   374→          </span>
   375→        )}
   376→      </div>
   377→      {diff.perProtocol.length > 0 && (
   378→        <div className="mt-2.5 border-t border-violet-500/15 pt-2.5">
   379→          <p className="mb-1.5 text-[10px] font-semibold uppercase tracking-wider text-slate-500">
   380→            Рост по протоколам — срез A → срез B
   381→          </p>
   382→          <div
   383→            className="flex h-1.5 w-full overflow-hidden rounded-full bg-slate-800"
   384→            role="img"
   385→            aria-label={`Дельты по протоколам между срезами: ${diff.perProtocol
   386→              .map(([p, n]) => `${p} +${n}`)
   387→              .join(", ")}`}
   388→          >
   389→            {diff.perProtocol.map(([proto, n]) => (
   390→              <div
   391→                key={proto}
   392→                className={`${protocolDot(proto)} transition-[flex-basis] duration-700`}
   393→                style={{ flexBasis: `${Math.max(2, (n / protoTotal) * 100)}%` }}
   394→                title={`${proto}: +${n}`}
   395→              />
   396→            ))}
   397→          </div>
   398→          <div className="mt-1.5 flex flex-wrap gap-x-3 gap-y-1">
   399→            {diff.perProtocol.map(([proto, n]) => (
   400→              <span
   401→                key={proto}
   402→                className="inline-flex items-center gap-1.5 font-mono text-[10px] text-slate-300"
   403→              >
   404→                <span className={`size-1.5 rounded-full ${protocolDot(proto)}`} aria-hidden />
   405→                {proto}
   406→                <span className="font-semibold text-violet-300 tabular-nums">+{fmtCompact(n)}</span>
   407→              </span>
   408→            ))}
   409→          </div>
   410→        </div>
   411→      )}
   412→    </div>
   413→  );
   414→}

   535→                  <Button
   536→                    size="sm"
   537→                    variant="ghost"

   549→                  <Loader2 className="size-4 animate-spin" aria-hidden /> Загрузка дампа…

   552→                <pre className="max-h-72 overflow-auto rounded-lg border border-slate-800 bg-slate-950 p-3 font-mono text-[10px] leading-relaxed text-emerald-300 custom-scroll">

   560→            <Loader2 className="size-4 animate-spin" aria-hidden /> Запрос GET /api/sniffer/packet/{pendingId}…
   561→          </div>
   562→        ) : null}
   563→      </SheetContent>
   564→    </Sheet>
   565→  );
   566→}
