   159→/** Подсветка совпадений поиска (без мутаций — чистый split по regex). */
   160→function Highlight({ text, query }: { text: string; query: string }) {
   161→  const q = query.trim();
   162→  if (!q) return <>{text}</>;
   163→  const escaped = q.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
   164→  let parts: string[];
   165→  try {
   166→    parts = text.split(new RegExp(`(${escaped})`, "ig"));
   167→  } catch {
   168→    return <>{text}</>;
   169→  }
   170→  const lower = q.toLowerCase();
   171→  return (
   172→    <>
   173→      {parts.map((part, i) =>
   174→        part.toLowerCase() === lower ? (
   175→          <mark key={i} className="rounded-[3px] bg-emerald-500/25 px-0.5 text-emerald-200">
   176→            {part}
   177→          </mark>
   178→        ) : (
   179→          <span key={i}>{part}</span>
   180→        )
   181→      )}
   182→    </>
   183→  );
   184→}

   214→function PacketDetailSheet({
   215→  packet,
   216→  pendingId,
   217→  open,
   218→  onOpenChange,
   219→  detail,
   220→  loading,
   221→}: {
   222→  packet: Packet | null;
   223→  pendingId: number | null;
   224→  open: boolean;
   225→  onOpenChange: (v: boolean) => void;
   226→  detail: { hexdump: string } | null;
   227→  loading: boolean;
   228→}) {
   229→  const [copied, setCopied] = useState(false);
   230→
   231→  const rows: [string, string][] = packet
   232→    ? [
   233→        ["ID", `#${packet.id}`],
   234→        ["Время", fmtTime(packet.ts)],
   235→        ["Направление", packet.direction],
   236→        ["Клиент", packet.client],
   237→        ["Канал", packet.portName],
   238→        ["Протокол", packet.protocol],
   239→        ["msg_type", packet.msgType ?? "—"],
   240→        ["method_type", packet.methodType ?? "—"],
   241→        ["method", packet.method ?? "—"],
   242→        ["cmd_type", packet.cmdType ?? "—"],
   243→        ["cmd_class / cmd", packet.cmdClass != null ? `${packet.cmdClass} / ${packet.cmd ?? "—"}` : "—"],
   244→        ["Размер", fmtBytes(packet.size)],
   245→        ["Валидность", packet.valid ? "✓ корректный" : "✗ повреждён/некорректен"],
   246→      ]
   247→    : [];
   248→
   249→  const hexdumpText = detail?.hexdump ?? packet?.hexPreview ?? "";
   250→
   251→  const handleCopy = async () => {
   252→    if (!hexdumpText) return;
   253→    try {
   254→      await navigator.clipboard.writeText(hexdumpText);
   255→      setCopied(true);
   256→      setTimeout(() => setCopied(false), 1500);
   257→    } catch {
   258→      /* clipboard недоступен */
   259→    }
   260→  };
   261→
   262→  return (
   263→    <Sheet open={open} onOpenChange={onOpenChange}>
   264→      <SheetContent
   265→        side="right"
   266→        className="w-full max-w-md overflow-y-auto border-slate-800 bg-slate-950 p-4 sm:max-w-lg custom-scroll"
   267→      >
   268→        <SheetHeader className="p-0 pb-2 text-left">
   269→          <SheetTitle className="text-base text-slate-100">
   270→            Пакет {packet ? `#${packet.id}` : pendingId ? `#${pendingId}` : ""}
   271→          </SheetTitle>
   272→          <SheetDescription className="text-xs text-slate-400">
   273→            {packet ? packet.summary : pendingId ? "Загрузка пакета из буфера сервера…" : ""}
   274→          </SheetDescription>
   275→        </SheetHeader>
   276→        {packet ? (
   277→          <div className="mt-2 space-y-3">
   278→            <div className="rounded-lg border border-slate-800">
   279→              <Table>
   280→                <TableBody>
   281→                  {rows.map(([k, v]) => (
   282→                    <TableRow key={k} className="border-slate-800/70">
   283→                      <TableCell className="py-1.5 pl-3 pr-2 font-mono text-[11px] text-slate-500">
   284→                        {k}
   285→                      </TableCell>
   286→                      <TableCell className="py-1.5 pr-3 font-mono text-xs text-slate-200">
   287→                        {v}
   288→                      </TableCell>
   289→                    </TableRow>
   290→                  ))}
   291→                </TableBody>
   292→              </Table>
   293→            </div>
   294→            <div>
   295→              <div className="mb-1 flex items-center justify-between">
   296→                <p className="text-[11px] font-semibold uppercase tracking-wider text-slate-500">
   297→                  Hexdump {loading ? "" : detail ? "(до 2 КБ)" : ""}
   298→                </p>
   299→                {!loading && hexdumpText && (
   300→                  <Button
   301→                    size="sm"
   302→                    variant="ghost"
   303→                    onClick={handleCopy}
   304→                    className="h-7 gap-1 px-2 text-[11px] text-slate-400 hover:bg-slate-800 hover:text-emerald-300"
   305→                    aria-label="Скопировать hexdump"
   306→                  >
   307→                    <Copy className="size-3" aria-hidden />
   308→                    {copied ? "скопировано" : "копировать"}
   309→                  </Button>
   310→                )}
   311→              </div>
   312→              {loading ? (
   313→                <div className="flex items-center gap-2 rounded-lg border border-slate-800 bg-slate-950 p-4 text-xs text-slate-400">
   314→                  <Loader2 className="size-4 animate-spin" aria-hidden /> Загрузка дампа…
   315→                </div>
   316→              ) : (
   317→                <pre className="max-h-72 overflow-auto rounded-lg border border-slate-800 bg-slate-950 p-3 font-mono text-[10px] leading-relaxed text-emerald-300 custom-scroll">
   318→                  {hexdumpText}
   319→                </pre>
   320→              )}
   321→            </div>
   322→          </div>
   323→        ) : pendingId ? (
   324→          <div className="mt-4 flex items-center gap-2 rounded-lg border border-slate-800 bg-slate-950 p-4 text-xs text-slate-400">
   325→            <Loader2 className="size-4 animate-spin" aria-hidden /> Запрос GET /api/sniffer/packet/{pendingId}…
   326→          </div>
   327→        ) : null}
   328→      </SheetContent>
   329→    </Sheet>
   330→  );
   331→}

   333→export function SnifferTab({ live }: SnifferTabProps) {
   334→  const { connected, packets, sessions, alerts, stats, series, scenario } = live;
   335→
   336→  // Фильтры
   337→  const [search, setSearch] = useState("");
   338→  const [protocol, setProtocol] = useState("all");
   339→  const [direction, setDirection] = useState("all");
   340→  const [minSize, setMinSize] = useState("");
   341→  const [maxSize, setMaxSize] = useState("");
   342→  const [paused, setPaused] = useState(false);
   343→  const [frozenPackets, setFrozenPackets] = useState<Packet[] | null>(null);
   344→  const [frozenTopId, setFrozenTopId] = useState(0);
   345→  const [muted, setMuted] = useState(false);
   346→  const [selected, setSelected] = useState<Packet | null>(null);
   347→  const [pendingId, setPendingId] = useState<number | null>(null);
   348→  const [sheetOpen, setSheetOpen] = useState(false);
   349→  const [detail, setDetail] = useState<{ hexdump: string } | null>(null);
   350→  const [detailLoading, setDetailLoading] = useState(false);
   351→  const [metric, setMetric] = useState<ChartMetric>("pps");
   352→  const [scenarioBusy, setScenarioBusy] = useState<ScenarioId | "stop" | null>(null);
   353→  const [nowTick, setNowTick] = useState(() => Date.now());
   354→  const [inspectorRaw, setInspectorRaw] = useState<Session | null>(null);
   355→  const [inspectorId, setInspectorId] = useState<string | null>(null);
   356→  const [soundOn, setSoundOn] = useState(false);
   357→  const [visibleCount, setVisibleCount] = useState(MAX_TABLE_ROWS);
   358→  const searchInputRef = useRef<HTMLInputElement>(null);
   359→  const tableCardRef = useRef<HTMLDivElement>(null);
   360→  const alertToastSeenRef = useRef<string | null>(null);
   361→  const soundSeenRef = useRef<string | null>(null);
   362→
   363→  // Тосты по живым алертам (если не в mute) — реагируем только на новые id
   364→  useEffect(() => {
   365→    if (muted || alerts.length === 0) return;
   366→    const latest = alerts[0];
   367→    if (alertToastSeenRef.current === latest.id) return;
   368→    const t = new Date(latest.createdAt).getTime();
   369→    if (Date.now() - t < 2500) {
   370→      alertToastSeenRef.current = latest.id;
   371→      if (latest.severity === "crit") {
   372→        toast.error(latest.rule, { description: latest.message, duration: 3500 });
   373→      } else {
   374→        toast.warning(latest.rule, { description: latest.message, duration: 3000 });
   375→      }
   376→    }
   377→  }, [alerts, muted]);
   378→
   379→  // Звук критических тревог (WebAudio, если включён тумблером)
   380→  useEffect(() => {
   381→    if (!soundOn || alerts.length === 0) return;
   382→    const latest = alerts[0];
   383→    if (latest.severity !== "crit") return;
   384→    if (soundSeenRef.current === latest.id) return;
   385→    const t = new Date(latest.createdAt).getTime();
   386→    if (Date.now() - t < 2500) {
   387→      soundSeenRef.current = latest.id;
   388→      playCritBeep();
   389→    }
   390→  }, [alerts, soundOn]);
   391→
   392→  // Сброс пагинации при смене фильтров
   393→  useEffect(() => {
   394→    setVisibleCount(MAX_TABLE_ROWS);
   395→  }, [search, protocol, direction, minSize, maxSize]);
   396→
   397→  // Тикаем часы для обратного отсчёта сценария
   398→  useEffect(() => {
   399→    if (!scenario) return;
   400→    const timer = setInterval(() => setNowTick(Date.now()), 1000);
   401→    return () => clearInterval(timer);
   402→  }, [scenario]);
   403→
   404→  const scenarioRemaining = useMemo(() => {
   405→    if (!scenario) return 0;
   406→    return Math.max(0, Math.ceil((new Date(scenario.endsAt).getTime() - nowTick) / 1000));
   407→  }, [scenario, nowTick]);
   408→
   409→  const handleScenario = async (id: ScenarioId | "stop") => {
   410→    setScenarioBusy(id);
   411→    try {
   412→      const res =
   413→        id === "stop"
   414→          ? await fetch("/api/sniffer/scenario", { method: "DELETE" })
   415→          : await fetch("/api/sniffer/scenario", {
   416→              method: "POST",
   417→              headers: { "Content-Type": "application/json" },
   418→              body: JSON.stringify({ id }),
   419→            });
   420→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   421→      if (id === "stop") {
   422→        toast("Сценарий остановлен", { duration: 2000 });
   423→      } else {
   424→        const def = SCENARIO_CATALOG.find((s) => s.id === id);
   425→        toast.success(`Сценарий «${def?.label}» запущен`, {
   426→          description: "Генератор трафика изменил профиль — следите за графиком и тревогами.",
   427→          duration: 3000,
   428→        });
   429→      }
   430→    } catch (e) {
   431→      toast.error("Не удалось изменить сценарий", {
   432→        description: e instanceof Error ? e.message : String(e),
   433→        duration: 3000,
   434→      });
   435→    } finally {
   436→      setScenarioBusy(null);
   437→    }
   438→  };
   439→
   440→  const handlePause = () => {
   441→    if (!paused) {
   442→      setFrozenPackets(packets);
   443→      setFrozenTopId(packets[0]?.id ?? 0);
   444→      setPaused(true);
   445→      toast.info("Таблица остановлена", {
   446→        description: "Приём пакетов продолжается в фоне (буфер 3000).",
   447→        duration: 2500,
   448→      });
   449→    } else {
   450→      setFrozenPackets(null);
   451→      setFrozenTopId(0);
   452→      setPaused(false);
   453→    }
   454→  };
   455→
   456→  // «Перейти к свежим» с плавающей пилюли — снять паузу и проскроллить к таблице
   457→  const jumpToFresh = () => {
   458→    setFrozenPackets(null);
   459→    setFrozenTopId(0);
   460→    setPaused(false);
   461→    tableCardRef.current?.scrollIntoView({ behavior: "smooth", block: "start" });
   462→  };
   463→
   464→  // Горячие клавиши вкладки: P — пауза, / — фокус на поиск
   465→  useEffect(() => {
   466→    const onKey = (e: KeyboardEvent) => {
   467→      const t = e.target as HTMLElement | null;
   468→      const typing =
   469→        !!t && (t.tagName === "INPUT" || t.tagName === "TEXTAREA" || t.tagName === "SELECT" || t.isContentEditable);
   470→      if (e.key.toLowerCase() === "p" || e.key === "з") {
   471→        if (typing || sheetOpen) return;
   472→        e.preventDefault();
   473→        handlePause();
   474→      } else if (e.key === "/") {
   475→        if (sheetOpen) return;
   476→        e.preventDefault();
   477→        searchInputRef.current?.focus();
   478→      }
   479→    };
   480→    window.addEventListener("keydown", onKey);
   481→    return () => window.removeEventListener("keydown", onKey);
   482→  }, [sheetOpen, packets, paused]);
   483→
   484→  const openPacketById = (id: number) => {
   485→    const local = packets.find((p) => p.id === id);
   486→    if (local) {
   487→      onRowClick(local);
   488→      return;
   489→    }
   490→    setPendingId(id);
   491→    setSelected(null);
   492→    setSheetOpen(true);
   493→    setDetail(null);
   494→    setDetailLoading(true);
   495→    fetch(`/api/sniffer/packet/${id}`)
   496→      .then((r) => (r.ok ? r.json() : Promise.reject(new Error(r.status === 404 ? "notfound" : String(r.status)))))
   497→      .then((d: { packet: Packet; hexdump: string }) => {
   498→        setSelected(d.packet);
   499→        setDetail({ hexdump: d.hexdump });
   500→      })
   501→      .catch(() => {
   502→        toast.error(`Пакет #${id} недоступен`, {
   503→          description: "Вытеснен из кольцевого буфера сниффера (5000 пакетов).",
   504→          duration: 3000,
   505→        });
   506→        setSheetOpen(false);
   507→      })
   508→      .finally(() => {
   509→        setPendingId(null);
   510→        setDetailLoading(false);
   511→      });
   512→  };
   513→
   514→  const filterByClient = (clientAddr: string) => {
   515→    const ip = clientAddr.split(":")[0];
   516→    setSearch(ip);
   517→    setPaused(false);
   518→    setFrozenPackets(null);
   519→    tableCardRef.current?.scrollIntoView({ behavior: "smooth", block: "start" });
   520→    toast(`Фильтр по клиенту ${ip}`, {
