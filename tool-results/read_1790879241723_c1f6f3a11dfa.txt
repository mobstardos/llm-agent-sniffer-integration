   189→   189→function Highlight({ text, query }: { text: string; query: string }) {
   190→   190→  const q = query.trim();
   191→   191→  if (!q) return <>{text}</>;
   192→   192→  const escaped = q.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
   193→   193→  let parts: string[];
   194→   194→  try {
   195→   195→    parts = text.split(new RegExp(`(${escaped})`, "ig"));
   196→   196→  } catch {
   197→   197→    return <>{text}</>;
   198→   198→  }
   199→   199→  const lower = q.toLowerCase();
   200→   200→  return (
   201→   201→    <>
   202→   202→      {parts.map((part, i) =>
   203→   203→        part.toLowerCase() === lower ? (
   204→   204→          <mark key={i} className="rounded-[3px] bg-emerald-500/25 px-0.5 text-emerald-200">
   205→   205→            {part}
   206→   206→          </mark>
   207→   207→        ) : (
   208→   208→          <span key={i}>{part}</span>
   209→   209→        )
   210→   210→      )}

   253→   253→/** Тренд KPI: среднее последних 60 с против предыдущих 60 с (из series). */
   254→   254→interface Trend {
   255→   255→  pct: number;
   256→   256→  dir: "up" | "down" | "flat";
   257→   257→}
   258→   258→
   259→   259→function computeTrend(values: number[]): Trend | null {
   260→   260→  if (values.length < 70) return null;
   261→   261→  const last = values.slice(-60);
   262→   262→  const prev = values.slice(-120, -60);
   263→   263→  if (prev.length < 30) return null;
   264→   264→  const avg = (arr: number[]) => arr.reduce((a, b) => a + b, 0) / arr.length;
   265→   265→  const now = avg(last);
   266→   266→  const before = avg(prev);
   267→   267→  if (before < 1e-9) return null;
   268→   268→  const pct = ((now - before) / before) * 100;
   269→   269→  if (Math.abs(pct) < 3) return { pct, dir: "flat" };
   270→   270→  return { pct, dir: pct > 0 ? "up" : "down" };
   271→   271→}

   309→   309→    const perProtocol: Record<string, number> = {};
   310→   310→    for (const [proto, now] of Object.entries(b.perProtocol)) {
   311→   311→      const delta = now - (a.perProtocol[proto] ?? 0);
   312→   312→      if (delta > 0) perProtocol[proto] = delta;
   313→   313→    }
   314→   314→    const durationSec = Math.max(
   315→   315→      1,
   316→   316→      Math.round((new Date(b.ts).getTime() - new Date(a.ts).getTime()) / 1000)
   317→   317→    );
   318→   318→    return {
   319→   319→      durationSec,
   320→   320→      packets: Math.max(0, b.totalPackets - a.totalPackets),
   321→   321→      bytes: Math.max(0, b.totalBytes - a.totalBytes),
   322→   322→      alerts: Math.max(0, b.alerts - a.alerts),
   323→   323→      invalid: Math.max(0, b.invalid - a.invalid),
   324→   324→      perProtocol: Object.entries(perProtocol).sort((x, y) => y[1] - x[1]),
   325→   325→    };
   326→   326→  }, [a, b]);
   327→   327→  const protoTotal = diff.perProtocol.reduce((acc, [, n]) => acc + n, 0);
