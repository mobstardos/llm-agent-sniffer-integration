   252→function pick<T>(arr: readonly T[]): T {
   253→  return arr[Math.floor(Math.random() * arr.length)];
   254→}
   255→
   256→function weightedPick<T>(entries: readonly [T, number][]): T {
   257→  const total = entries.reduce((s, [, w]) => s + w, 0);
   258→  let r = Math.random() * total;
   259→  for (const [v, w] of entries) {
   260→    r -= w;
   261→    if (r <= 0) return v;
   262→  }
   263→  return entries[entries.length - 1][0];
   264→}

   270→/** Классический hexdump в стиле xxd/hd (как parsers/hexdump.py) */
   271→export function hexdump(data: Uint8Array, maxBytes?: number): string {
   272→  const limit = maxBytes ? Math.min(data.length, maxBytes) : data.length;
   273→  const lines: string[] = [];
   274→  const HEX = "0123456789abcdef";
   275→  for (let off = 0; off < limit; off += 16) {
   276→    const chunk = data.subarray(off, Math.min(off + 16, limit));
   277→    let hex = "";
   278→    let ascii = "";
   279→    for (let i = 0; i < 16; i++) {
   280→      if (i < chunk.length) {
   281→        const b = chunk[i];
   282→        hex += HEX[(b >> 4) & 0xf] + HEX[b & 0xf] + (i === 7 ? "  " : " ");
   283→        ascii += b >= 0x20 && b < 0x7f ? String.fromCharCode(b) : ".";
   284→      } else {
   285→        hex += "   ";
   286→        ascii += " ";
   287→      }
   288→    }
   289→    lines.push(
   290→      off.toString(16).padStart(8, "0") + "  " + hex + " |" + ascii + "|"
   291→    );
   292→  }
   293→  if (data.length > limit) {
   294→    lines.push(
   295→      `... (${data.length - limit} байт обрезано; всего ${data.length} Б)`
   296→    );
   297→  }
   298→  return lines.join("\n");
   299→}

   314→function draftRemoteServer(
   315→  client: string,
   316→  direction: Direction,
   317→  force?: { c: number; d: number }
   318→): PacketDraft {
   319→  const cmd = force ?? pick(REMOTE_CMD_POOL);
   320→  const key = `${cmd.c}:${cmd.d}`;
   321→  const cmdType = CMD_TYPES[key] ?? "UNKNOWN";
   322→  const session = randInt(100000, 999999);
   323→  const txid = randInt(1000, 99999);
   324→  const valid = Math.random() > 0.03;
   325→
   326→  // Реальный заголовок: len(4LE) + MAGIC AA AA 00 00 + версия
   327→  const header = new Uint8Array([
   328→    randInt(0, 255),
   329→    randInt(0, 255),
   330→    randInt(0, 255),
   331→    randInt(0, 255),
   332→    0xaa,
   333→    0xaa,
   334→    0x00,
   335→    0x00,
   336→    0x02,
   337→  ]);
   338→  let bodyStr = `RequestCmdClass${cmd.c}RequestCmd${cmd.d}SessionNum${session}TransactionID${txid}`;
   339→  if (cmdType === "DB_REQUEST" && Math.random() < 0.07) {
   340→    bodyStr += ` query=SELECT * FROM ${pick(KNOWN_TABLES)}; -- error: table locked`;
   341→  } else if (cmdType === "DB_REQUEST") {
   342→    bodyStr += ` query=SELECT id,name FROM ${pick(KNOWN_TABLES)} LIMIT 50`;
   343→  }
   344→  if (cmdType === "SEND_TANKS_STATE") {
   345→    bodyStr += ` tanks=[{tank:1,fuel:AI-95,level:${randInt(1000, 5000)}}]`;
   346→  }
   347→  bodyStr += ` ${pick(KNOWN_TABLES)}`;
   348→  const body = new TextEncoder().encode(bodyStr.slice(0, 900));
   349→  const full = new Uint8Array(header.length + body.length);
   350→  full.set(header, 0);
   351→  full.set(body, header.length);
   352→
   353→  const parts: string[] = [
   354→    `[RemoteServer TCP] REMOTE_SERVER ${cmdType} (C=${cmd.c},D=${cmd.d})`,
   355→  ];
   356→  if (Math.random() < 0.7) parts.push(`S=${session}`);
   357→  if (Math.random() < 0.5) parts.push(`TX=${txid}`);
   358→  if (Math.random() < 0.25) parts.push("zlib");
   359→
   360→  return {
   361→    packet: {
   362→      direction,
   363→      client,
   364→      portName: "RemoteServer TCP",
   365→      protocol: "REMOTE_SERVER",
   366→      cmdType,
   367→      cmdClass: cmd.c,
   368→      cmd: cmd.d,
   369→      size: full.length + randInt(60, 8000),
   370→      valid,
   371→      summary: parts.join(" "),
   372→    },
   373→    body: full,
   374→  };
   375→}

   377→function draftThrift(
   378→  client: string,
   379→  direction: Direction,
   380→  seqRef: { seq: number }
   381→): PacketDraft {
   382→  const method = pick(THRIFT_METHODS);
   383→  const roll = Math.random();
   384→  // В обычном режиме EXCEPTION ~6%; в сценарии «Thrift-шторм» — ~30%
   385→  const exceptionRate = S.scenario?.id === "thrift_storm" ? 0.3 : 0.06;
   386→  const msgType =
   387→    roll >= exceptionRate
   388→      ? direction === "TX"
   389→        ? "CALL"
   390→        : "REPLY"
   391→      : "EXCEPTION";
   392→  seqRef.seq += 1;
   393→  const seq = seqRef.seq;
   394→  const valid = msgType !== "EXCEPTION";
   395→
   396→  let payload = "";
   397→  if (msgType === "CALL") {
   398→    payload = `${method} seq=${seq} args={stationId:${randInt(1, 250)}, strict:true}`;
   399→  } else if (msgType === "REPLY") {
   400→    payload = `${method} seq=${seq} success=true result={items:${randInt(0, 120)}}`;
   401→  } else {
   402→    payload = `${method} seq=${seq} TApplicationException: Internal error processing ${method} (tank ${randInt(1, 8)} timeout)`;
   403→  }
   404→  const body = new TextEncoder().encode(payload);
   405→
   406→  return {
   407→    packet: {
   408→      direction,
   409→      client,
   410→      portName: "Thrift RPC",
   411→      protocol: "THRIFT",
   412→      msgType,
   413→      method,
   414→      size: body.length + randInt(80, 6000),
   415→      valid,
   416→      summary: `[Thrift RPC] THRIFT ${msgType} ${method} (seq=${seq})`,
   417→    },
   418→    body,
   419→  };
   420→}

   422→function draftHttp(client: string, direction: Direction): PacketDraft {
   423→  const method = Math.random() < 0.6 ? "GET" : "POST";
   424→  const path = pick(HTTP_PATHS);
   425→  const status = pick([200, 200, 200, 204, 400, 404, 500]);
   426→  const isResp = direction === "RX";
   427→  const firstLine = isResp
   428→    ? `HTTP/1.1 ${status} ${status === 200 ? "OK" : status === 204 ? "No Content" : status === 400 ? "Bad Request" : status === 404 ? "Not Found" : "Internal Server Error"}`
   429→    : `${method} ${path} HTTP/1.1`;
   430→  const body = new TextEncoder().encode(
   431→    `${firstLine}\r\nHost: azs-core.local\r\nContent-Type: application/json\r\n\r\n{"ok":${status < 400}}`
   432→  );
   433→  return {
   434→    packet: {
   435→      direction,
   436→      client,
   437→      portName: "HTTP API",
   438→      protocol: "HTTP",
   439→      msgType: isResp ? "RESPONSE" : "REQUEST",
   440→      method: isResp ? `HTTP ${status}` : method,
   441→      size: body.length + randInt(120, 12000),
   442→      valid: status < 500,
   443→      summary: `[HTTP API] HTTP ${isResp ? `RESPONSE ${status}` : `REQUEST ${method} ${path}`}`,
   444→    },
   445→    body,
   446→  };
   447→}

   471→function draftModbus(client: string, direction: Direction): PacketDraft {
   472→  const func = pick([3, 4, 6]);
   473→  const unit = randInt(1, 24);
   474→  const addr = randInt(0x0000, 0x0fff);
   475→  const qty = randInt(1, 32);
   476→  const value = randInt(0, 65535);
   477→  const isResp = direction === "RX";
   478→  const body =
   479→    isResp && func === 6
   480→      ? new Uint8Array([
   481→          unit,
   482→          func,
   483→          (addr >> 8) & 0xff,
   484→          addr & 0xff,
   485→          (value >> 8) & 0xff,
   486→          value & 0xff,
   487→        ])
   488→      : new Uint8Array([
   489→          unit,
   490→          func,
   491→          (addr >> 8) & 0xff,
   492→          addr & 0xff,
   493→          (qty >> 8) & 0xff,
   494→          qty & 0xff,
   495→        ]);
   496→  const msgType = isResp ? "RESPONSE" : "REQUEST";
   497→  const detail =
   498→    func === 3 || func === 4
   499→      ? `read ${isResp ? `${qty} reg` : "holding"} addr=0x${addr
   500→          .toString(16)
   501→          .padStart(4, "0")}`
   502→      : `write addr=0x${addr.toString(16).padStart(4, "0")} value=${value}`;
   503→  return {
   504→    packet: {
   505→      direction,
   506→      client,
   507→      portName: "Modbus TCP",
   508→      protocol: "MODBUS",
   509→      msgType,
   510→      methodType: `FUNC_${func}`,
   511→      method: `func=${func}`,
   512→      size: body.length + randInt(0, 300),
   513→      valid: Math.random() > 0.05,
   514→      summary: `[Modbus TCP] MODBUS ${msgType} func=${func} ${detail} unit=${unit}`,
   515→    },
   516→    body,
   517→  };
   518→}

   658→function openSession(): void {
   659→  const active = [...S.sessions.values()].filter((s) => s.state === "active");
   660→  if (active.length >= 7) return;
   661→  // Сначала добираем отсутствующие каналы, чтобы демо показывало все протоколы,
   662→  // затем взвешенный выбор как в config.json
   663→  const present = new Set(active.map((s) => s.portName));
   664→  const missing = (Object.keys(PROTOCOL_BY_PORT) as PortName[]).filter(
   665→    (p) => !present.has(p)
   666→  );
   667→  let portName: PortName;
   668→  if (missing.length > 0 && (active.length < 5 || Math.random() < 0.5)) {
   669→    portName = missing[Math.floor(Math.random() * missing.length)];
   670→  } else {
   671→    portName = weightedPick<PortName>([
   672→      ["RemoteServer TCP", 45],
   673→      ["Thrift RPC", 30],
   674→      ["HTTP API", 15],
   675→      ["Modbus TCP", 8],
   676→      ["RAW", 2],
   677→    ]);
   678→  }
   679→  // Подбираем свободный адрес (ip:port), до 5 попыток
   680→  for (let attempt = 0; attempt < 5; attempt++) {
   681→    const client = randomClientAddr();
   682→    const key = `${client}|${portName}`;
   683→    if (S.sessions.has(key)) continue;
   684→    const session: Session = {
   685→      id: newId(),
   686→      client,
   687→      portName,
   688→      protocol: PROTOCOL_BY_PORT[portName],
   689→      startedAt: new Date().toISOString(),
   690→      txPackets: 0,
   691→      rxPackets: 0,
   692→      txBytes: 0,
   693→      rxBytes: 0,
   694→      state: "active",
   695→    };
   696→    S.sessions.set(key, session);
   697→    S.totalSessions += 1;
   698→    emit({ type: "session", payload: session });
   699→    return;
   700→  }
   701→}

   703→function maintainSessions(): void {
   704→  const active = [...S.sessions.values()].filter((s) => s.state === "active");
   705→  if (active.length < 3 && Math.random() < 0.7) openSession();
   706→  else if (active.length < 7 && Math.random() < 0.3) openSession();
   707→  if (active.length > 3 && Math.random() < 0.15) {
   708→    const victim = pick(active);
   709→    victim.state = "closing";
   710→    emit({ type: "session", payload: victim });
   711→    const key = [...S.sessions.entries()].find(
   712→      ([, s]) => s.id === victim.id
   713→    )?.[0];
   714→    if (key) {
   715→      setTimeout(() => {
   716→        const s = S.sessions.get(key);
   717→        if (s && s.state === "closing") {
   718→          s.state = "closed";
   719→          emit({ type: "session", payload: s });
   720→          setTimeout(() => {
   721→            if (S.sessions.get(key)?.state === "closed") S.sessions.delete(key);
   722→          }, 5000);
   723→        }
   724→      }, 3000);
   725→    }
   726→  }
   727→  // страховка от разрастания карты
   728→  if (S.sessions.size > 60) {
   729→    for (const [k, s] of S.sessions) {
   730→      if (s.state === "closed") S.sessions.delete(k);
   731→      if (S.sessions.size <= 40) break;
   732→    }
   733→  }
   734→}

   736→function trackSession(packet: Packet): void {
   737→  const key = `${packet.client}|${packet.portName}`;
   738→  let session = S.sessions.get(key);
   739→  if (!session) {
   740→    // Прямой ingest без сессии (fallback) — учитываем в счетчиках,
   741→    // но не плодим новые сессии сверх лимита
   742→    const activeCount = [...S.sessions.values()].filter(
   743→      (s) => s.state === "active"
   744→    ).length;
   745→    if (activeCount >= 8) return;
   746→    session = {
   747→      id: newId(),
   748→      client: packet.client,
   749→      portName: packet.portName,
   750→      protocol: packet.protocol,
   751→      startedAt: packet.ts,
   752→      txPackets: 0,
   753→      rxPackets: 0,
   754→      txBytes: 0,
   755→      rxBytes: 0,
   756→      state: "active",
   757→    };
   758→    S.sessions.set(key, session);
   759→    S.totalSessions += 1;
   760→    emit({ type: "session", payload: session });
   761→  }
   762→  if (packet.direction === "TX") {
   763→    session.txPackets += 1;
   764→    session.txBytes += packet.size;
   765→  } else {
   766→    session.rxPackets += 1;
   767→    session.rxBytes += packet.size;
   768→  }
   769→}

   809→function checkAlertRules(packet: Packet): void {
   810→  const now = Date.now();
   811→
   812→  const cooldownFor = (rule: string): number => {
   813→    // В рамках сценариев частота тревог повышается — иначе шторм не виден
   814→    if (S.scenario?.id === "thrift_storm" && rule === "Thrift EXCEPTION") return 6_000;
   815→    if (S.scenario?.id === "giant_attack" && rule === "Гигантский пакет") return 7_000;
   816→    if (S.scenario?.id === "rollback_loop" && rule === "Откат безналичной оплаты") return 12_000;
   817→    return ALERT_COOLDOWN_MS;
   818→  };
   819→
   820→  const fire = (rule: string, cooldownMs: number): boolean => {
   821→    const last = S.alertCooldown.get(rule) ?? 0;
   822→    if (now - last < cooldownMs) return false;
   823→    S.alertCooldown.set(rule, now);
   824→    return true;
   825→  };
   826→
   827→  // 1. Thrift EXCEPTION — crit
   828→  if (packet.protocol === "THRIFT" && packet.msgType === "EXCEPTION") {
   829→    if (fire("Thrift EXCEPTION", cooldownFor("Thrift EXCEPTION"))) {
   830→      raiseAlert(
   831→        "Thrift EXCEPTION",
   832→        "crit",
   833→        `Thrift-исключение в ${packet.method ?? "методе"}: TApplicationException (пакет #${packet.id})`,
   834→        packet
   835→      );
   836→    }
   837→  }
   838→
   839→  // 2. Откат безналичной оплаты — crit, порог 2 за 60с
   840→  if (packet.methodType === "CASHLESS_ROLLBACK") {
   841→    S.rollbackTimes.push(now);
   842→    S.rollbackTimes = S.rollbackTimes.filter(
   843→      (t) => now - t <= ROLLBACK_WINDOW_MS
   844→    );
   845→    if (S.rollbackTimes.length >= ROLLBACK_THRESHOLD) {
   846→      if (fire("Откат безналичной оплаты", cooldownFor("Откат безналичной оплаты"))) {
   847→        S.rollbackTimes = [];
   848→        raiseAlert(
   849→          "Откат безналичной оплаты",
   850→          "crit",
   851→          `Обнаружено ${ROLLBACK_THRESHOLD} отката CASHLESS_ROLLBACK за ${ROLLBACK_WINDOW_MS / 1000}с — требуется проверка кассовой логики`,
   852→          packet
   853→        );
   854→      }
   855→    }
   856→  }
   857→
   858→  // 3. Гигантский пакет — warn
   859→  if (packet.size > GIANT_SIZE) {
   860→    if (fire("Гигантский пакет", cooldownFor("Гигантский пакет"))) {
   861→      raiseAlert(
   862→        "Гигантский пакет",
   863→        "warn",
   864→        `Размер пакета ${(packet.size / 1_000_000).toFixed(2)} МБ превышает порог 1 000 000 Б`,
   865→        packet
   866→      );
   867→    }
   868→  }
   869→
   870→  // 4. Ошибка БД в запросе — warn (DB_REQUEST + «error» в полезной нагрузке)
   871→  if (
   872→    packet.cmdType === "DB_REQUEST" &&
   873→    packet.summary.toLowerCase().includes("error")
   874→  ) {
   875→    if (fire("Ошибка БД в запросе", ALERT_COOLDOWN_MS)) {
   876→      raiseAlert(
   877→        "Ошибка БД в запросе",
   878→        "warn",
   879→        `DB_REQUEST содержит «error» — вероятно, сбой выполнения SQL (пакет #${packet.id})`,
   880→        packet
   881→      );
   882→    }
   883→  }
   884→}

   888→function makePacket(scenarioId?: ScenarioId): Packet {
   889→  // Пакет привязан к одной из активных сессий (как в реальном прокси):
   890→  // стабильный клиент ip:port на время жизни сессии, протокол = канал сессии
   891→  const active = [...S.sessions.values()].filter((s) => s.state === "active");
   892→  const session = active.length > 0 ? pick(active) : null;
   893→  const client = session ? session.client : randomClientAddr();
   894→  const direction: Direction = Math.random() < 0.55 ? "TX" : "RX";
   895→  let protocol: Protocol = session
   896→    ? session.protocol
   897→    : weightedPick<Protocol>([
   898→        ["REMOTE_SERVER", 45],
   899→        ["THRIFT", 30],
   900→        ["HTTP", 10],
   901→        ["JSON", 8],
   902→        ["MODBUS", 5],
   903→        ["RAW", 2],
   904→      ]);
   905→
   906→  // Профиль сценария: смещаем протокол/направление
   907→  if (scenarioId === "azs_burst") {
   908→    protocol = Math.random() < 0.8 ? "REMOTE_SERVER" : protocol;
   909→  } else if (scenarioId === "thrift_storm") {
   910→    protocol = "THRIFT";
   911→  } else if (scenarioId === "giant_attack") {
   912→    // протокол любой, размер зададим ниже
   913→  }
   914→
   915→  let draft: PacketDraft;
   916→  switch (protocol) {
   917→    case "REMOTE_SERVER":
   918→      draft = draftRemoteServer(client, direction);
   919→      break;
   920→    case "THRIFT": {
   921→      const seqRef = { seq: S.thriftSeq };
   922→      draft = draftThrift(client, direction, seqRef);
   923→      S.thriftSeq = seqRef.seq;
   924→      break;
   925→    }
   926→    case "HTTP":
   927→      draft = draftHttp(client, direction);
   928→      break;
   929→    case "JSON":
   930→      draft = draftJson(client, direction);
   931→      break;
   932→    case "MODBUS":
   933→      draft = draftModbus(client, direction);
   934→      break;
   935→    default:
   936→      draft = draftRaw(client, direction);
   937→  }
   938→
   939→  // редкие гигантские пакеты (~0.5%) или сценарий «Гигантские пакеты»
   940→  if (scenarioId === "giant_attack") {
   941→    draft.packet.size = randInt(1_100_000, 1_500_000);
   942→  } else if (Math.random() < 0.005) {
   943→    draft.packet.size = randInt(1_100_000, 1_400_000);
   944→  }
   945→
   946→  return {
   947→    id: S.nextPacketId++,
   948→    ts: new Date().toISOString(),
   949→    hexPreview: hexdump(draft.body, 128),
   950→    ...draft.packet,
   951→  };
   952→}

   954→function ingest(packet: Packet): void {
   955→  S.packets.push(packet);
   956→  if (S.packets.length > PACKETS_CAP) {
   957→    S.packets.splice(0, S.packets.length - PACKETS_CAP);
   958→  }
   959→  S.totalPackets += 1;
   960→  S.totalBytes += packet.size;
   961→  S.secondPackets += 1;
   962→  S.secondBytes += packet.size;
   963→  if (!packet.valid) S.invalidPackets += 1;
   964→  S.perProtocol[packet.protocol] = (S.perProtocol[packet.protocol] ?? 0) + 1;
   965→
   966→  const methodKey =
   967→    packet.methodType ?? packet.method ?? packet.msgType ?? packet.protocol;
   968→  S.methodCounter.set(methodKey, (S.methodCounter.get(methodKey) ?? 0) + 1);
   969→  S.clientCounter.set(packet.client, (S.clientCounter.get(packet.client) ?? 0) + 1);
   970→
   971→  trackSession(packet);
   972→  checkAlertRules(packet);
   973→  emit({ type: "packet", payload: packet });
   974→}

   976→function tick(): void {
   977→  // Сценарии: множитель пакетов/профиль протокола
   978→  const sc = S.scenario;
   979→  if (sc && Date.now() > new Date(sc.endsAt).getTime()) {
   980→    // Сценарий самозавершился по таймеру
   981→    S.scenario = null;
   982→    emit({ type: "scenario", payload: null });
   983→  }
   984→
   985→  const scenarioId = S.scenario?.id;
   986→  const multiplier = scenarioId === "azs_burst" ? 4 : 1;
   987→  for (let i = 0; i < multiplier; i++) {
   988→    ingest(makePacket(scenarioId));
   989→  }
   990→
   991→  if (scenarioId === "rollback_loop") {
   992→    // Серии откатов на каждом тике — правило срабатывает нон-стоп
   993→    const active = [...S.sessions.values()].filter((s) => s.state === "active");
   994→    const rollbackClient = active.length > 0 ? pick(active).client : randomClientAddr();
   995→    for (let i = 0; i < 2; i++) {
   996→      const d = draftRemoteServer(rollbackClient, "TX", { c: 200, d: 111 });
   997→      ingest({
   998→        id: S.nextPacketId++,
   999→        ts: new Date().toISOString(),
  1000→        hexPreview: hexdump(d.body, 128),
  1001→        ...d.packet,
  1002→      });
  1003→    }
  1004→  }
  1005→
  1006→  // Редкие всплески откатов безналичной оплаты (~2% тиков) вне сценария:
  1007→  // 2-3 пакета CASHLESS_ROLLBACK подряд — срабатывает правило алерта
  1008→  if (!scenarioId && Math.random() < 0.02) {
  1009→    const burst = randInt(2, 3);
  1010→    const active = [...S.sessions.values()].filter((s) => s.state === "active");
  1011→    const rollbackClient =
  1012→      active.length > 0
  1013→        ? pick(active).client
  1014→        : randomClientAddr();
  1015→    for (let i = 0; i < burst; i++) {
  1016→      const d = draftRemoteServer(rollbackClient, "TX", { c: 200, d: 111 });
  1017→      ingest({
  1018→        id: S.nextPacketId++,
  1019→        ts: new Date().toISOString(),
  1020→        hexPreview: hexdump(d.body, 128),
  1021→        ...d.packet,
  1022→      });
  1023→    }
  1024→  }
  1025→}

  1047→    [...m.entries()]
  1048→      .sort((a, b) => b[1] - a[1])
  1049→      .slice(0, n)
  1050→      .map(([name, count]) => ({ name, count }));

  1053→    startedAt: S.startedAt,
  1054→    uptimeSec: Math.floor(
  1055→      (Date.now() - new Date(S.startedAt).getTime()) / 1000
  1056→    ),
  1057→    totalPackets: S.totalPackets,
  1058→    totalBytes: S.totalBytes,
  1059→    activeSessions: [...S.sessions.values()].filter(
  1060→      (s) => s.state === "active"
  1061→    ).length,
  1062→    totalSessions: S.totalSessions,
  1063→    alertsCount: S.alerts.length,
  1064→    perProtocol: { ...S.perProtocol },
  1065→    topMethods: top(S.methodCounter, 8),
  1066→    topClients: top(S.clientCounter, 8),
  1067→    lastPps: S.lastPps,
  1068→    lastBps: S.lastBps,
  1069→    invalidPackets: S.invalidPackets,
  1070→  };

  1123→export function getScenario(): ScenarioState | null {
  1124→  if (!S.scenario) return null;

  1150→export function getFullHexdump(packet: Packet): string {
  1151→  // Для демо: синтезируем до 2КБ данных на основе packet.summary
  1152→  const header = new TextEncoder().encode(packet.summary);
  1153→  const total = Math.min(packet.size, 2048);
  1154→  const buf = new Uint8Array(total);
  1155→  buf.set(header.subarray(0, Math.min(header.length, total)), 0);
  1156→  // магия RemoteServer в начале, если применимо
  1157→  if (packet.protocol === "REMOTE_SERVER") {
  1158→    buf[4] = 0xaa;
  1159→    buf[5] = 0xaa;
  1160→    buf[6] = 0x00;
  1161→    buf[7] = 0x00;
  1162→  }
  1163→  for (let i = Math.min(header.length, total); i < total; i++) {
  1164→    buf[i] = randInt(0, 255);
  1165→  }
  1166→  return hexdump(buf);
  1167→}
